-- Creation of Database
CREATE DATABASE university1;
-- Display available databases
SHOW DATABASES;
-- Select the database to work with
USE university1;

-- Creation of Table
CREATE TABLE student (
     roll_num INT PRIMARY KEY ,
     name VARCHAR(50) ,
     dept VARCHAR(3)
);
-- Display tables present in the selected database
SHOW TABLES;

-- Insertion of first student record
INSERT INTO student VALUES ( 1, "Vijay Shekhar", "CSE" );
-- Insertion of second student record
INSERT INTO student VALUES ( 2, "Ramesh Kumar", "ECE" );
-- Insertion of third student record
INSERT INTO student VALUES ( 3, "Ajeet Raahi", "CSE" );

-- Updation of existing student record
-- WHERE ensures that only roll_num = 2 is updated
UPDATE student SET name = "Mahima Manohari" WHERE roll_num = 2;

-- Alter Table: Adding a new column
ALTER TABLE student ADD COLUMN ( email VARCHAR(50) );

-- Display all columns and all rows
SELECT * FROM student;

-- SELECT specific columns and rename them using AS
SELECT roll_num AS "Roll Number", name AS "Student Name",
     dept AS "Department", email AS "Email"
     FROM student;

-- DISTINCT returns unique department values
SELECT DISTINCT dept FROM student;

-- WHERE filters records based on a condition
-- LIKE '%r' finds names ending with the letter 'r'
SELECT name FROM student WHERE name LIKE '%r';

-- Insertion of a record including the newly added email column
INSERT INTO student VALUES ( 4, "Manisha Vidyarthi", "CSE", "manisha@bhu.ac.in" );

-- WHERE + IS NULL: Find students without an email
SELECT * FROM student WHERE email IS NULL;

-- WHERE + IS NOT NULL: Find students having an email
SELECT * FROM student WHERE email IS NOT NULL;
